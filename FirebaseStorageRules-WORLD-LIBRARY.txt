// 3DX World Library Storage additions (already merged into FirebaseStorageRules-full-updated.rules)

match /world-library/{userId}/{worldId}/images/{fileName} {
  allow read: if true;
  allow create, update: if request.auth != null && request.auth.uid == userId && request.resource.size <= 15 * 1024 * 1024 && request.resource.contentType.matches('image/(jpeg|png|webp|gif)');
  allow delete: if request.auth != null && request.auth.uid == userId;
}

match /world-library/{userId}/{worldId}/world/{fileName} {
  allow read: if true;
  allow create, update: if request.auth != null && request.auth.uid == userId && fileName.matches('.*[.][wW][oO][rR][lL][dD]$') && request.resource.size <= 250 * 1024 * 1024 && request.resource.contentType == 'application/octet-stream';
  allow delete: if request.auth != null && request.auth.uid == userId;
}

match /world-library-private/{userId}/{worldId}/world/{fileName} {
  allow read: if false;
  allow create, update: if request.auth != null && request.auth.uid == userId && fileName.matches('.*[.][wW][oO][rR][lL][dD]$') && request.resource.size <= 250 * 1024 * 1024 && request.resource.contentType == 'application/octet-stream';
  allow delete: if request.auth != null && request.auth.uid == userId;
}
