//
// Do not replace your full Storage rules with this file. The site already uses
// Firebase Storage for avatars, banners, chat images and video uploads.
//
// Security model:
// - Anyone can DOWNLOAD a published World Library file/image.
// - A signed-in account can only upload inside its OWN uid folder.
// - Realtime Database rules separately restrict /worldLibrary metadata to the
//   Mayflower Studios owner/admin accounts, so random users cannot publish a
//   world into the visible library even if they know this storage path.

match /world-library/{userId}/{worldId}/images/{fileName} {
  allow read: if true;
  allow create, update: if request.auth != null
    && request.auth.uid == userId
    && request.resource.size <= 15 * 1024 * 1024
    && request.resource.contentType.matches('image/(jpeg|png|webp|gif)');
  allow delete: if request.auth != null && request.auth.uid == userId;
}

match /world-library/{userId}/{worldId}/world/{fileName} {
  allow read: if true;
  allow create, update: if request.auth != null
    && request.auth.uid == userId
    && request.resource.size <= 250 * 1024 * 1024
    && request.resource.contentType == 'application/octet-stream';
  allow delete: if request.auth != null && request.auth.uid == userId;
}
